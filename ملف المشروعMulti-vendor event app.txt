from django.db import models
from django.contrib.auth.models import AbstractUser

# نوع المستخدم
class User(AbstractUser):
    USER_TYPE_CHOICES = (
        ('client', 'عميل'),
        ('vendor', 'متعاهد'),
        ('agent', 'وكيل مديرية'),
        ('admin', 'إدارة'),
    )
    user_type = models.CharField(max_length=10, choices=USER_TYPE_CHOICES)

# جغرافي
class City(models.Model):
    name = models.CharField(max_length=100)  # اسم المدينة

class District(models.Model):
    city = models.ForeignKey(City, on_delete=models.CASCADE)
    name = models.CharField(max_length=100)  # مديرية

class Neighborhood(models.Model):
    district = models.ForeignKey(District, on_delete=models.CASCADE)
    name = models.CharField(max_length=100)  # حي

class Block(models.Model):
    neighborhood = models.ForeignKey(Neighborhood, on_delete=models.CASCADE)
    name = models.CharField(max_length=100)  # حارة

# وكيل مديرية مرتبط بالمديرية
class Agent(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    district = models.OneToOneField(District, on_delete=models.CASCADE)

# المتعاهد مرتبط بحي وحارة
class Vendor(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    district = models.ForeignKey(District, on_delete=models.CASCADE)
    neighborhood = models.ForeignKey(Neighborhood, on_delete=models.CASCADE)
    block = models.ForeignKey(Block, on_delete=models.CASCADE)
    # أنواع الخدمات التي يقدمها
    service_types = models.ManyToManyField('ServiceType')

# نوع المناسبة
class EventType(models.Model):
    name = models.CharField(max_length=100)  # مثل: عرس، تخرج، عيد ميلاد

# نوع الخدمة
class ServiceType(models.Model):
    name = models.CharField(max_length=100)  # مثل: كوش، تصوير، ديكور

# الخدمات أو المنتجات التي يقدمها المتعاهد
class Service(models.Model):
    vendor = models.ForeignKey(Vendor, on_delete=models.CASCADE)
    event_type = models.ForeignKey(EventType, on_delete=models.CASCADE)
    service_type = models.ForeignKey(ServiceType, on_delete=models.CASCADE)
    title = models.CharField(max_length=200)
    description = models.TextField()
    price = models.DecimalField(max_digits=10, decimal_places=2)
    is_rent = models.BooleanField(default=False)  # هل الخدمة للايجار
    is_purchase = models.BooleanField(default=True)  # هل الخدمة للشراء
    available = models.BooleanField(default=True)

# الحجز
class Booking(models.Model):
    client = models.ForeignKey(User, on_delete=models.CASCADE, limit_choices_to={'user_type': 'client'})
    service = models.ForeignKey(Service, on_delete=models.CASCADE)
    event_date = models.DateField()
    quantity = models.PositiveIntegerField(default=1)
    
    PAYMENT_TYPE_CHOICES = (
        ('full', 'الدفع كامل'),
        ('deposit', 'مبلغ مقدم'),
        ('insurance', 'مبلغ تأمين'),
    )
    payment_type = models.CharField(max_length=10, choices=PAYMENT_TYPE_CHOICES)
    amount_paid = models.DecimalField(max_digits=10, decimal_places=2)
    is_paid = models.BooleanField(default=False)

    created_at = models.DateTimeField(auto_now_add=True)

views.py
from rest_framework import viewsets, permissions
from .models import Service, Booking
from .serializers import ServiceSerializer, BookingSerializer

class ServiceViewSet(viewsets.ModelViewSet):
    queryset = Service.objects.filter(available=True)
    serializer_class = ServiceSerializer
    permission_classes = [permissions.AllowAny]

class BookingViewSet(viewsets.ModelViewSet):
    serializer_class = BookingSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        return Booking.objects.filter(client=self.request.user)

    def perform_create(self, serializer):
        serializer.save(client=self.request.user)

توجه إلى مجلد المشروع عبر الأمر:

cd مسار_المجلد_المفكوك
إنشاء بيئة افتراضية وتنشيطها:

python -m venv env
على Windows:

.\env\Scripts\activate
على Linux / Mac:

source env/bin/activate
تثبيت المكتبات المطلوبة:

pip install -r requirements.txt
تشغيل الترحيلات (migrations) لإنشاء قواعد البيانات:

python manage.py migrate
إنشاء مستخدم مسؤول (Admin) للدخول للوحة الإدارة:

python manage.py createsuperuser
أدخل اسم المستخدم، البريد، وكلمة المرور.

تشغيل السيرفر محليًا:

python manage.py runserver
فتح المتصفح وكتابة العنوان:

http://127.0.0.1:8000/
لوحة الإدارة (Admin Panel):
http://127.0.0.1:8000/admin/
سجل دخولك بمستخدم الـ admin الذي أنشأته.

واجهة الـ API (مثال):
http://127.0.0.1:8000/api/services/

4. ماذا بعد؟
تستطيع اختبار إضافة مستخدمين (متعاهدين، عملاء، وكلاء) من لوحة الإدارة.

تستطيع إدخال خدمات جديدة وربطها بالمناسبات.

تجربة الحجز عبر API.

أساعدك بإضافة واجهة مستخدم بعد ذلك.

سأجهز لك نسخة كاملة مبسطة لتطبيق "هلا المتعاهد الشامل" بلغة Python باستخدام Django وDjango REST Framework، مع جميع الملفات الضرورية لتشغيله محلياً.

سأرفق لك ملفات المشروع مضغوطة (zip) مع شرح كامل لتثبيته وتشغيله.

خطوات التحضير والتشغيل بعد تحميل الملف:
ثبت بايثون 3.10 أو أحدث من python.org (تأكد من اختيار “Add to PATH” أثناء التثبيت).

فك ضغط المجلد في مكان مناسب.

افتح الطرفية (Terminal أو CMD) وانتقل لمجلد المشروع.

نفذ الأمر لإنشاء بيئة افتراضية وتفعيلها:

على Windows:

python -m venv env
.\env\Scripts\activate
على Linux / Mac:

python3 -m venv env
source env/bin/activate
ثبت المتطلبات:

pip install -r requirements.txt
شغل الترحيلات لإنشاء قاعدة البيانات:

python manage.py migrate
أنشئ مستخدم مسؤول (للدخول للوحة الإدارة):

python manage.py createsuperuser
شغل الخادم:

python manage.py runserver
افتح المتصفح على:

http://127.0.0.1:8000/admin/ للدخول للوحة الإدارة

http://127.0.0.1:8000/api/services/ لعرض الخدمات (مثال API)
